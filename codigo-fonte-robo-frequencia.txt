def processar_tabela_com_beautifulsoup(html_content):
    soup = BeautifulSoup(html_content, 'html.parser')
    tabela = soup.find('table', class_='po-table')
    if not tabela: raise ValueError("ERRO: Nenhuma tabela de frequência foi encontrada na página.")
    thead = tabela.find('thead')
    if not thead: raise ValueError("ERRO: Cabeçalho da tabela não encontrado.")
    linhas_cabecalho = thead.find_all('tr')
    cabecalho_final = []
    if len(linhas_cabecalho) > 1:
        datas_th = linhas_cabecalho[0].find_all('th', colspan=True)
        lista_datas_expandida = [d.text.strip() for d in datas_th for _ in range(int(d.get('colspan',1)))]
        horarios_th = linhas_cabecalho[1].find_all('th', class_='subtitle')
        lista_horarios = [th.text.strip() for th in horarios_th]
        cabecalho_final = [f"{d} {h}" for d,h in zip(lista_datas_expandida, lista_horarios)]
    elif len(linhas_cabecalho) == 1:
        headers_th = linhas_cabecalho[0].find_all('th', class_=lambda c: c is not None and 'sticky-col' not in c)
        cabecalho_final = [th.text.strip() for th in headers_th]
    else: raise ValueError("ERRO: Estrutura de cabeçalho não reconhecida.")
    dados_alunos = []
    linhas_alunos = tabela.find('tbody').find_all('tr', class_='tr-data')
    for linha in linhas_alunos:
        info_aluno = {}
        try:
            info_aluno['RA'] = linha.find('td', {'data-cy': 'ra'}).text.strip()
            info_aluno['Nome'] = linha.find('td', {'data-cy': 'name'}).text.strip()
            celulas_presenca = linha.find_all('td', {'data-cy': lambda x: x and x.startswith('x')})
            for i, celula in enumerate(celulas_presenca):
                if i < len(cabecalho_final):
                    coluna_nome = cabecalho_final[i]
                    switch = celula.find('po-switch')
                    info_aluno[coluna_nome] = 'FALTA' if switch and 'switch-checked' in switch.get('class', []) else 'Presente'
            dados_alunos.append(info_aluno)
        except AttributeError: continue
    return pd.DataFrame(dados_alunos)
 
def gerar_corpo_email(lista_alunos_criticos):
    if not lista_alunos_criticos: return ""
    alunos_desligamento = [a for a in lista_alunos_criticos if len(a['Dias_Falta']) >= 15]
    alunos_acompanhamento = [a for a in lista_alunos_criticos if 2 <= len(a['Dias_Falta']) < 15]
    data_hora = datetime.now().strftime("%d/%m/%Y às %H:%M")
    email = f"Assunto: Relatório de Faltas Consecutivas - {data_hora}\n\n"
    email += "Prezados,\n\nSegue o relatório de alunos com 2 ou mais dias de faltas totais consecutivas para acompanhamento.\n"
    if alunos_desligamento:
        email += "\n" + "="*70 + "\nALERTA DE RISCO DE DESLIGAMENTO (15+ faltas)\n" + "="*70 + "\n\n"
        for a in alunos_desligamento: email += f"Nome: {a['Nome']}\nRA: {a['RA']}\nTotal de Faltas: {len(a['Dias_Falta'])}\nDias: {', '.join(a['Dias_Falta'])}\n---\n"
    if alunos_acompanhamento:
        email += "\n" + "="*70 + "\nALUNOS PARA ACOMPANHAMENTO (2 a 14 faltas)\n" + "="*70 + "\n\n"
        for a in alunos_acompanhamento: email += f"Nome: {a['Nome']}\nRA: {a['RA']}\nTotal de Faltas: {len(a['Dias_Falta'])}\nDias: {', '.join(a['Dias_Falta'])}\n---\n"
    email += "\n\nAtenciosamente,\nAutomação de Relatórios."
    return email
 
def calcular_periodo_pesquisa(dias_uteis_a_voltar):
    hoje = datetime.now()
    dia_da_semana_hoje = hoje.weekday()
    if dia_da_semana_hoje > 3:
        offset = dia_da_semana_hoje - 3
        data_fim = hoje - timedelta(days=offset)
    else: data_fim = hoje
    dias_contados = 0
    data_inicio = data_fim
    while dias_contados < dias_uteis_a_voltar - 1:
        data_inicio -= timedelta(days=1)
        if data_inicio.weekday() < 4: dias_contados += 1
    return data_inicio, data_fim

def analisar_frequencia(df_frequencia):
    alunos_criticos = []
    colunas_datas = [col for col in df_frequencia.columns if '/' in col]
    if not colunas_datas: return []
 
    ano_atual = str(datetime.now().year)
    datas_unicas_str = sorted(list(set([col.split(' ')[0] for col in colunas_datas])))
    try: datas_unicas_obj = [datetime.strptime(f"{d}/{ano_atual}", "%d/%m/%Y") for d in datas_unicas_str]
    except ValueError:
        ano_anterior = str(datetime.now().year - 1)
        datas_unicas_obj = [datetime.strptime(f"{d}/{ano_anterior}", "%d/%m/%Y") for d in datas_unicas_str]
   
    if not datas_unicas_obj: return []
   
    datas_ordenadas_reverso = sorted(datas_unicas_obj, reverse=True)
 
    for _, aluno in df_frequencia.iterrows():
        faltas_consecutivas_atuais = 0
        dias_da_falta = []
       
        for dia_obj in datas_ordenadas_reverso:
            if dia_obj.weekday() > 3: continue
 
            dia_str = dia_obj.strftime("%d/%m")
            aulas_do_dia = [c for c in colunas_datas if c.startswith(dia_str)]
            status_aulas = [aluno.get(aula) for aula in aulas_do_dia]
           
            # --- LÓGICA DE FALTA TOTAL CORRIGIDA ---
            # 1. Filtra apenas os status que não são vazios (ignora 'nan')
            status_validos = [s for s in status_aulas if pd.notna(s)]
           
            # 2. Verifica se a lista de status válidos não está vazia e se TODOS são 'FALTA'
            foi_falta_total = False
            if status_validos and all(s == 'FALTA' for s in status_validos):
                foi_falta_total = True
           
            if foi_falta_total:
                faltas_consecutivas_atuais += 1
                dias_da_falta.append(dia_obj)
            else:
                break # Quebra a sequência se encontrar um dia com presença ou sem aulas válidas
       
        if faltas_consecutivas_atuais >= 2:
            dias_formatados = [d.strftime("%d/%m/%Y") for d in sorted(dias_da_falta)]
            alunos_criticos.append({'RA': aluno['RA'], 'Nome': aluno['Nome'], 'Dias_Falta': dias_formatados})
           
    return alunos_criticos

def executar_automacao(usuario, senha, links_frequencia_str, diretorio_saida, rodar_invisivel, atualizar_status_callback):
    LOGIN_URL = "https://portaldocente.fiemg.com.br/FrameHTML/Web/App/Edu/PortalDoProfessor/#/login"
    chrome_options = Options()
    chrome_options.add_experimental_option('excludeSwitches', ['enable-logging'])
    chrome_options.add_argument('--start-maximized')
    chrome_options.add_argument("--incognito")
    if rodar_invisivel:
        chrome_options.add_argument('--headless'); chrome_options.add_argument('--disable-gpu')
    driver = None
    try:
        lista_urls = [url.strip() for url in links_frequencia_str.strip().splitlines() if url.strip()]
        lista_dataframes = []
        data_inicial_pesquisa, data_final_pesquisa = calcular_periodo_pesquisa(15)
        data_inicio_str = data_inicial_pesquisa.strftime("%d/%m/%Y")
        data_fim_str = data_final_pesquisa.strftime("%d/%m/%Y")
        for i, url in enumerate(lista_urls):
            atualizar_status_callback(f"Iniciando link {i+1}/{len(lista_urls)}...", "yellow")
            driver = webdriver.Chrome(options=chrome_options)
            wait = WebDriverWait(driver, 30)
            atualizar_status_callback(f"({i+1}/{len(lista_urls)}) Fazendo login...", "yellow")
            driver.get(LOGIN_URL)
            wait.until(EC.visibility_of_element_located((By.CSS_SELECTOR, 'input[name="login"]'))).send_keys(usuario.strip())
            wait.until(EC.visibility_of_element_located((By.CSS_SELECTOR, 'input[name="password"]'))).send_keys(senha.strip())
            time.sleep(1)
            botao_login = wait.until(EC.element_to_be_clickable((By.XPATH, "//button[.//span[contains(text(), 'Entrar')]]")))
            driver.execute_script("arguments[0].click();", botao_login)
            time.sleep(5)
            atualizar_status_callback(f"({i+1}/{len(lista_urls)}) Acessando frequência...", "yellow")
            driver.get(url)
            time.sleep(5)
            atualizar_status_callback(f"Filtrando de {data_inicio_str} a {data_fim_str}...", "yellow")
            try:
                campo_data_inicio = wait.until(EC.visibility_of_element_located((By.CSS_SELECTOR, 'input.mat-start-date')))
                driver.execute_script("arguments[0].value = arguments[1];", campo_data_inicio, data_inicio_str)
                driver.execute_script("arguments[0].dispatchEvent(new Event('input', { bubbles: true }));", campo_data_inicio)
                campo_data_fim = wait.until(EC.visibility_of_element_located((By.CSS_SELECTOR, 'input.mat-end-date')))
                driver.execute_script("arguments[0].value = arguments[1];", campo_data_fim, data_fim_str)
                driver.execute_script("arguments[0].dispatchEvent(new Event('input', { bubbles: true }));", campo_data_fim)
                driver.execute_script("arguments[0].dispatchEvent(new Event('change', { bubbles: true }));", campo_data_fim)
                botao_pesquisar = wait.until(EC.element_to_be_clickable((By.XPATH, "//button[.//span[normalize-space()='Pesquisar']]")))
                botao_pesquisar.click()
                time.sleep(5)
            except Exception as e:
                raise Exception(f"Erro ao tentar filtrar o período de datas: {e}")
            SELETOR_TABELA = "//table[.//th//span[contains(text(),'Ra')] and .//th//span[contains(text(),'Nome')]]"
            wait.until(EC.presence_of_element_located((By.XPATH, SELETOR_TABELA)))
            time.sleep(3)
            html_content = driver.find_element(By.XPATH, SELETOR_TABELA).get_attribute('outerHTML')
            df_parcial = processar_tabela_com_beautifulsoup(html_content)
            if not df_parcial.empty:
                lista_dataframes.append(df_parcial)
            atualizar_status_callback(f"({i+1}/{len(lista_urls)}) Link processado. Fechando navegador...", "yellow")
            driver.quit()
            driver = None
        if not lista_dataframes:
            atualizar_status_callback("Nenhum dado de aluno encontrado nos links.", "orange")
            return
        atualizar_status_callback("Juntando dados e iniciando análise final...", "yellow")
        df_completo = pd.concat(lista_dataframes, ignore_index=True).drop_duplicates(subset=['RA']).reset_index(drop=True)
        lista_final_alertas = analisar_frequencia(df_completo)
        if lista_final_alertas:
            corpo_do_email = gerar_corpo_email(lista_final_alertas)
            caminho_txt = os.path.join(diretorio_saida, 'email_alerta_faltas.txt')
            with open(caminho_txt, 'w', encoding='utf-8') as f: f.write(corpo_do_email)
            atualizar_status_callback(f"Sucesso! 'email_alerta_faltas.txt' salvo!", "lightgreen")
        else:
            atualizar_status_callback("Análise concluída. Nenhum aluno em condição de alerta.", "orange")
    except Exception as e:
        if driver: driver.save_screenshot(os.path.join(diretorio_saida, 'screenshot_erro.png'))
        raise Exception(f"{e}\nScreenshot salvo.")
    finally:
        if driver: driver.quit()

    def selecionar_pasta_saida(self):
        pasta_selecionada = filedialog.askdirectory()
        if pasta_selecionada:
            self.entry_pasta.delete(0, "end")
            self.entry_pasta.insert(0, pasta_selecionada)
 
    def iniciar_thread_automacao(self):
        thread = threading.Thread(target=self.executar_logica_automacao, daemon=True)
        thread.start()
 
    def executar_logica_automacao(self):
        self.botao_iniciar.configure(state="disabled", text="Executando...")
       
        usuario = self.entry_usuario.get()
        senha = self.entry_senha.get()
        links_frequencia = self.textbox_links.get("1.0", "end-1c")
        pasta_saida = self.entry_pasta.get()
        rodar_invisivel = True if self.switch_background_var.get() == "on" else False
 
        if not all([usuario, senha, links_frequencia, pasta_saida]):
            self.atualizar_status("ERRO: Todos os campos são obrigatórios.", "red")
            self.botao_iniciar.configure(state="normal", text="Iniciar Análise")
            return
 
        self.salvar_credenciais()
 
        try:
            executar_automacao(usuario, senha, links_frequencia, pasta_saida, rodar_invisivel, self.atualizar_status)
        except Exception as e:
            self.atualizar_status(f"ERRO INESPERADO: {e}", "red")
 
        self.botao_iniciar.configure(state="normal", text="Iniciar Análise")
 
    def atualizar_status(self, mensagem, cor):
        self.label_status.configure(text=mensagem, text_color=cor)
 
    def salvar_credenciais(self):
        arquivo = "credenciais.json"
        if self.check_salvar_var.get() == "on":
            with open(arquivo, 'w') as f: json.dump({"usuario": self.entry_usuario.get(), "senha": self.entry_senha.get()}, f)
        elif os.path.exists(arquivo): os.remove(arquivo)
   
    def carregar_credenciais(self):
        arquivo = "credenciais.json"
        if os.path.exists(arquivo):
            with open(arquivo, 'r') as f:
                dados = json.load(f)
                self.entry_usuario.insert(0, dados.get("usuario", ""))
                self.entry_senha.insert(0, dados.get("senha", ""))
